```ts
import { normalizeProps, type PropTypes } from "@zag-js/vue"
import {
  ListVirtualizer,
  type ListVirtualizerOptions,
} from "@zag-js/virtualizer"
import {
  onScopeDispose,
  shallowRef,
  toValue,
  triggerRef,
  useId,
  watchEffect,
  type MaybeRefOrGetter,
} from "vue"

type Options = Omit<ListVirtualizerOptions, "id" | "normalizeProps">

export function useListVirtualizer(options: MaybeRefOrGetter<Options>) {
  const instance = new ListVirtualizer<PropTypes>({
    ...toValue(options),
    id: useId(),
    normalizeProps,
  })

  // The same instance, re-triggered on every change so templates re-render
  const virtualizer = shallowRef(instance)
  const unsubscribe = instance.subscribe(() => triggerRef(virtualizer))

  // Unchanged options and new callbacks don't re-render
  watchEffect(() => instance.updateOptions(toValue(options)), { flush: "pre" })

  let current: Element | null = null
  const ref = (element: Element | object | null) => {
    if (!(element instanceof HTMLElement) || element === current) return
    current = element
    instance.init(element)
  }

  onScopeDispose(() => {
    unsubscribe()
    instance.destroy()
  })

  return { virtualizer, ref }
}
```
