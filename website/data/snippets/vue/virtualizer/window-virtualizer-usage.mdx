```html
<script setup>
  const props = defineProps({
    items: { type: Array, required: true },
  })

  const { virtualizer, ref: rootRef } = useWindowVirtualizer(() => ({
    count: props.items.length,
    estimatedItemSize: 48,
    initialRect: { width: 1024, height: 720 },
    indexToKey: (index) => props.items[index].id,
  }))
</script>

<template>
  <div :ref="rootRef" v-bind="virtualizer.getRootProps()">
    <div v-bind="virtualizer.getContentProps()">
      <div
        v-for="virtualItem in virtualizer.getVirtualItems()"
        :key="virtualItem.key"
        v-bind="virtualizer.getItemProps(virtualItem)"
      >
        {{ props.items[virtualItem.index].label }}
      </div>
    </div>
  </div>
</template>
```
