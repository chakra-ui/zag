```html
<script setup>
  const props = defineProps({
    items: { type: Array, required: true },
  })

  const { virtualizer, init } = useWindowVirtualizer({
    count: props.items.length,
    estimatedSize: () => 48,
    overscan: 8,
    initialRect: { width: 1024, height: 720 },
    indexToKey: (index) => props.items[index].id,
  })

  virtualizer.updateOptions({ count: props.items.length })
</script>

<template>
  <div
    :ref="init"
    v-bind="virtualizer.getContainerAriaAttrs()"
    :style="virtualizer.getContainerStyle()"
  >
    <div :style="virtualizer.getContentStyle()">
      <div
        v-for="virtualItem in virtualizer.getVirtualItems()"
        :key="virtualItem.key"
        :ref="virtualItem.measureElement"
        v-bind="virtualizer.getItemAriaAttrs(virtualItem.index)"
        :style="virtualizer.getItemStyle(virtualItem)"
      >
        {{ props.items[virtualItem.index].label }}
      </div>
    </div>
  </div>
</template>
```
