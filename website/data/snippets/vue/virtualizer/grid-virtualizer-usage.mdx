```html
<script setup>
  const props = defineProps({
    rowCount: { type: Number, required: true },
    columnCount: { type: Number, required: true },
    getCell: { type: Function, required: true },
  })

  const { virtualizer, ref: rootRef } = useGridVirtualizer(() => ({
    rowCount: props.rowCount,
    columnCount: props.columnCount,
    estimatedRowSize: () => 40,
    estimatedColumnSize: () => 120,
  }))
</script>

<template>
  <div :ref="rootRef" v-bind="virtualizer.getRootProps()" class="grid">
    <div v-bind="virtualizer.getContentProps()">
      <div
        v-for="row in virtualizer.getVirtualRows()"
        :key="row.row"
        v-bind="virtualizer.getRowProps(row)"
      >
        <div
          v-for="column in row.columns"
          :key="column.column"
          v-bind="virtualizer.getCellProps(row, column)"
        >
          {{ props.getCell(row.row, column.column) }}
        </div>
      </div>
    </div>
  </div>
</template>
```
