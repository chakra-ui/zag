```jsx
import * as meter from "@zag-js/meter"
import { normalizeProps, useMachine } from "@zag-js/solid"
import { createMemo, createUniqueId } from "solid-js"

function Meter() {
  const service = useMachine(meter.machine, () => ({
    id: createUniqueId(),
    defaultValue: 70,
    low: 60,
    high: 85,
    optimum: 10,
  }))

  const api = createMemo(() => meter.connect(service, normalizeProps))

  return (
    <div {...api().getRootProps()}>
      <div {...api().getLabelProps()}>Storage used</div>
      <div {...api().getValueTextProps()}>{api().valueAsString}</div>
      <div {...api().getTrackProps()}>
        <div {...api().getIndicatorProps()} />
      </div>
    </div>
  )
}
```
