```tsx
import {
  normalizeProps,
  type PropTypes,
  useSyncExternalStore,
} from "@zag-js/solid"
import {
  WindowVirtualizer,
  type WindowVirtualizerOptions,
} from "@zag-js/virtualizer"
import {
  type Accessor,
  createEffect,
  createMemo,
  createUniqueId,
  onCleanup,
  onSettled,
  untrack,
} from "solid-js"

type Options = Omit<WindowVirtualizerOptions, "id" | "normalizeProps">

export function useWindowVirtualizer(options: Options | Accessor<Options>) {
  const read = () => (typeof options === "function" ? options() : options)
  const instance = new WindowVirtualizer<PropTypes>({
    ...untrack(read),
    id: createUniqueId(),
    normalizeProps,
  })
  const snapshot = useSyncExternalStore(
    instance.subscribe,
    instance.getSnapshot,
  )

  // Unchanged options and new callbacks don't re-render
  createEffect(read, (next) => instance.updateOptions(next))

  // The same instance, re-read by dependents on every change
  const virtualizer = createMemo(() => (snapshot(), instance), {
    equals: false,
  })

  let rootElement: HTMLElement | null = null
  let mounted = false

  onSettled(() => {
    mounted = true
    if (rootElement) instance.init(rootElement)
  })

  onCleanup(() => instance.destroy())

  const ref = (element: HTMLElement | null) => {
    rootElement = element
    if (mounted && element) instance.init(element)
  }

  return { virtualizer, ref }
}
```
