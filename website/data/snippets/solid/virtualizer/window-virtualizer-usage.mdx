```tsx
import { For } from "solid-js"

function WindowList(props: { items: { id: string; label: string }[] }) {
  const { virtualizer, ref } = useWindowVirtualizer(() => ({
    count: props.items.length,
    estimatedItemSize: 48,
    initialRect: { width: 1024, height: 720 },
    indexToKey: (index) => props.items[index].id,
  }))

  return (
    <div ref={ref} {...virtualizer().getRootProps()}>
      <div {...virtualizer().getContentProps()}>
        <For keyed={false} each={virtualizer().getVirtualItems()}>
          {(virtualItem) => (
            <div {...virtualizer().getItemProps(virtualItem())}>
              {props.items[virtualItem().index].label}
            </div>
          )}
        </For>
      </div>
    </div>
  )
}
```
