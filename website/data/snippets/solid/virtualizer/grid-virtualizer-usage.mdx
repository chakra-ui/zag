```tsx
import { For } from "solid-js"

function Grid(props: {
  rowCount: number
  columnCount: number
  getCell: (row: number, column: number) => string
}) {
  const { virtualizer, ref } = useGridVirtualizer(() => ({
    rowCount: props.rowCount,
    columnCount: props.columnCount,
    estimatedRowSize: () => 40,
    estimatedColumnSize: () => 120,
  }))

  return (
    <div ref={ref} {...virtualizer().getRootProps()} class="grid">
      <div {...virtualizer().getContentProps()}>
        <For keyed={false} each={virtualizer().getVirtualRows()}>
          {(row) => (
            <div {...virtualizer().getRowProps(row())}>
              <For keyed={false} each={row().columns}>
                {(column) => (
                  <div {...virtualizer().getCellProps(row(), column())}>
                    {props.getCell(row().row, column().column)}
                  </div>
                )}
              </For>
            </div>
          )}
        </For>
      </div>
    </div>
  )
}
```
