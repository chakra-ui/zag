```svelte
<script lang="ts">
  let {
    rowCount,
    columnCount,
    getCell,
  }: {
    rowCount: number
    columnCount: number
    getCell: (row: number, column: number) => string
  } = $props()

  const id = $props.id()
  const vz = useGridVirtualizer(() => ({
    id,
    rowCount,
    columnCount,
    estimatedRowSize: () => 40,
    estimatedColumnSize: () => 120,
  }))
</script>

<div {@attach vz.init} {...vz.virtualizer.getRootProps()} class="grid">
  <div {...vz.virtualizer.getContentProps()}>
    {#each vz.virtualizer.getVirtualRows() as row (row.row)}
      <div {...vz.virtualizer.getRowProps(row)}>
        {#each row.columns as column (column.column)}
          <div {...vz.virtualizer.getCellProps(row, column)}>
            {getCell(row.row, column.column)}
          </div>
        {/each}
      </div>
    {/each}
  </div>
</div>
```
