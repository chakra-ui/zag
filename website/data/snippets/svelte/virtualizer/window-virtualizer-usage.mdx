```svelte
<script lang="ts">
  let { items }: { items: { id: string; label: string }[] } = $props()

  const id = $props.id()
  const vz = useWindowVirtualizer(() => ({
    id,
    count: items.length,
    estimatedItemSize: 48,
    initialRect: { width: 1024, height: 720 },
    indexToKey: (index) => items[index].id,
  }))
</script>

<div {@attach vz.init} {...vz.virtualizer.getRootProps()}>
  <div {...vz.virtualizer.getContentProps()}>
    {#each vz.virtualizer.getVirtualItems() as virtualItem (virtualItem.key)}
      <div {...vz.virtualizer.getItemProps(virtualItem)}>
        {items[virtualItem.index].label}
      </div>
    {/each}
  </div>
</div>
```
