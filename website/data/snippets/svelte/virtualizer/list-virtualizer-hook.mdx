```ts
import { normalizeProps, type PropTypes } from "@zag-js/svelte"
import {
  ListVirtualizer,
  type ListVirtualizerOptions,
} from "@zag-js/virtualizer"
import { onDestroy, untrack } from "svelte"

// Pass `id` from `$props.id()` so server-rendered markup matches
type Options = Omit<ListVirtualizerOptions, "normalizeProps"> & { id: string }

export function useListVirtualizer(options: () => Options) {
  const virtualizer = untrack(
    () => new ListVirtualizer<PropTypes>({ ...options(), normalizeProps }),
  )

  let version = $state(0)
  const unsubscribe = virtualizer.subscribe(() => version++)

  // Unchanged options and new callbacks don't re-render
  $effect.pre(() => {
    const next = options()
    untrack(() => virtualizer.updateOptions(next))
  })

  onDestroy(() => {
    unsubscribe()
    virtualizer.destroy()
  })

  return {
    // Read in templates to re-render when the virtualizer updates
    get virtualizer() {
      void version
      return virtualizer
    },
    // An attachment: `{@attach vz.init}`
    init: (element: HTMLElement) => virtualizer.init(element),
  }
}
```
