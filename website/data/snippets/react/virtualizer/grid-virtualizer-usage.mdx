function Grid(props: {
  rowCount: number
  columnCount: number
  getCell: (row: number, column: number) => string
}) {
  const { rowCount, columnCount, getCell } = props
  const { virtualizer, ref } = useGridVirtualizer({
    rowCount,
    columnCount,
    estimatedRowSize: () => 40,
    estimatedColumnSize: () => 120,
  })

  return (
    <div ref={ref} {...virtualizer.getRootProps()} className="grid">
      <div {...virtualizer.getContentProps()}>
        {virtualizer.getVirtualRows().map((row) => (
          <div key={row.row} {...virtualizer.getRowProps(row)}>
            {row.columns.map((column) => (
              <div
                key={column.column}
                {...virtualizer.getCellProps(row, column)}
              >
                {getCell(row.row, column.column)}
              </div>
            ))}
          </div>
        ))}
      </div>
    </div>
  )
}
