```tsx
function List({ items }: { items: { id: string; label: string }[] }) {
  const { virtualizer, ref } = useListVirtualizer({
    count: items.length,
    estimatedItemSize: 48,
    indexToKey: (index) => items[index].id,
  })

  return (
    <div ref={ref} {...virtualizer.getRootProps()} className="list">
      <div {...virtualizer.getContentProps()}>
        {virtualizer.getVirtualItems().map((virtualItem) => (
          <div key={virtualItem.key} {...virtualizer.getItemProps(virtualItem)}>
            {items[virtualItem.index].label}
          </div>
        ))}
      </div>
    </div>
  )
}
```
