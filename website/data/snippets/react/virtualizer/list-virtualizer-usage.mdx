```tsx
function List({ items }: { items: { id: string; label: string }[] }) {
  const { virtualizer, ref } = useListVirtualizer({
    count: items.length,
    estimatedSize: () => 48,
    overscan: 5,
    indexToKey: (index) => items[index].id,
  })

  virtualizer.updateOptions({ count: items.length })

  return (
    <div
      ref={ref}
      onScroll={virtualizer.handleScroll}
      {...virtualizer.getContainerAriaAttrs()}
      style={{ ...virtualizer.getContainerStyle(), height: 400 }}
    >
      <div style={virtualizer.getContentStyle()}>
        {virtualizer.getVirtualItems().map((virtualItem) => {
          const item = items[virtualItem.index]
          return (
            <div
              key={virtualItem.key}
              data-index={virtualItem.index}
              ref={virtualItem.measureElement}
              {...virtualizer.getItemAriaAttrs(virtualItem.index)}
              style={virtualizer.getItemStyle(virtualItem)}
            >
              {item.label}
            </div>
          )
        })}
      </div>
    </div>
  )
}
```
