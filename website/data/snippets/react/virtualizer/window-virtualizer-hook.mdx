```tsx
import { normalizeProps, type PropTypes } from "@zag-js/react"
import {
  WindowVirtualizer,
  type WindowVirtualizerOptions,
} from "@zag-js/virtualizer"
import {
  useCallback,
  useEffect,
  useId,
  useLayoutEffect,
  useState,
  useSyncExternalStore,
} from "react"

type Options = Omit<WindowVirtualizerOptions, "id" | "normalizeProps">

export function useWindowVirtualizer(options: Options) {
  const id = useId()
  const [virtualizer] = useState(
    () => new WindowVirtualizer<PropTypes>({ ...options, id, normalizeProps }),
  )
  useSyncExternalStore(
    virtualizer.subscribe,
    virtualizer.getSnapshot,
    virtualizer.getSnapshot,
  )

  // Pass options fresh on every render; unchanged ones don't re-render
  useLayoutEffect(() => virtualizer.updateOptions(options))

  const ref = useCallback(
    (element: HTMLElement | null) => {
      if (element) virtualizer.init(element)
    },
    [virtualizer],
  )

  useEffect(() => () => virtualizer.destroy(), [virtualizer])

  return { virtualizer, ref }
}
```
